// Book Search

let bookSearch = document.getElementById("bookSearch");

if(bookSearch){

bookSearch.addEventListener("keyup", function(){

let value = this.value.toLowerCase();

let rows = document.querySelectorAll("#bookTable tr");


rows.forEach(function(row,index){

if(index === 0) return;


let data = row.innerText.toLowerCase();


if(data.includes(value)){

row.style.display="";

}
else{

row.style.display="none";

}

});


});

}

function addBook(){

let name = document.getElementById("bookName").value;

let author = document.getElementById("authorName").value;

let category = document.getElementById("category").value;

let quantity = document.getElementById("quantity").value;


if(name=="" || author=="" || category==""){

alert("Please fill all details");

return;

}


let book = {

name:name,

author:author,

category:category,

quantity:quantity

};


let books = JSON.parse(localStorage.getItem("books")) || [];


books.push(book);


localStorage.setItem("books",JSON.stringify(books));


alert("Book Added Successfully");


window.location.href="books.html";


}

window.onload=function(){

}

function issueBook(){

    let student = document.getElementById("student").value;
    let book = document.getElementById("book").value;
    let date = document.getElementById("date").value;

    if(student=="" || book=="" || date==""){
        alert("Please fill all fields");
        return;
    }

    let table = document.getElementById("issueTable");

    let row = table.insertRow(-1);

    row.insertCell(0).innerHTML = table.rows.length - 1;
    row.insertCell(1).innerHTML = student;
    row.insertCell(2).innerHTML = book;
    row.insertCell(3).innerHTML = date;
    row.insertCell(4).innerHTML = "Issued";

    document.getElementById("student").value="";
    document.getElementById("book").value="";
    document.getElementById("date").value="";

    alert("Book Issued Successfully");
}

function filterIssuedBooks() {

    let selectedDate = document.getElementById("date").value;
    let table = document.getElementById("issueTable");

    for (let i = 1; i < table.rows.length; i++) {

        let issueDate = table.rows[i].cells[3].innerText.trim();

        if (selectedDate === "" || issueDate === selectedDate) {
            table.rows[i].style.display = "";
        } else {
            table.rows[i].style.display = "none";
        }
    }
}
